USE bdControlOrdenes;
SELECT TOP 20
    s.name + '.' + t.name AS tabla,
    SUM(CASE WHEN ps.index_id IN (0,1) THEN ps.row_count ELSE 0 END) AS filas,
    CAST(SUM(ps.reserved_page_count) * 8.0 / 1024 AS DECIMAL(12,1)) AS reservado_MB,
    CAST(SUM(ps.used_page_count) * 8.0 / 1024 AS DECIMAL(12,1)) AS usado_MB
FROM sys.dm_db_partition_stats ps
JOIN sys.tables t  ON t.object_id = ps.object_id
JOIN sys.schemas s ON s.schema_id = t.schema_id
GROUP BY s.name, t.name
ORDER BY SUM(ps.reserved_page_count) DESC;



SELECT name, type_desc,
       CAST(size * 8.0 / 1024 AS DECIMAL(12,1)) AS tamano_MB,
       CAST(FILEPROPERTY(name, 'SpaceUsed') * 8.0 / 1024 AS DECIMAL(12,1)) AS usado_MB
FROM sys.database_files;


DBCC SHRINKFILE (N'bdControlOrdenes', 15000);   -- MB objetivo: usado actual + holgura
DBCC SHRINKFILE (N'bdControlOrdenes_log', 256); -- si el log está grande
